from sage.all import *
import random

def hdr(title):
    print("=" * 72)
    print(title)
    print("=" * 72)

RESULTS = []

def record(name, condition, detail=""):
    RESULTS.append((name, bool(condition), detail))
    status = "PASS" if condition else "FAIL"
    print(f"[{status}] {name}")
    if detail:
        print(f"       {detail}")
    return condition


def module11_rank_modular():
    hdr("Module 11: Rank theorem for larger B, S (modular)")
    p = 2305843009213693951  # 2^61 - 1, a Mersenne prime
    print(f"  Prime p = {p}")
    print(f"  p ≈ 2^61, so Schwartz-Zippel false-positive prob ≤ S/p < 1e-16")
    print()

    def R_mod_p(B, S, G_val):
        F = GF(p)
        G = F(G_val)

        max_i = 2*B + S + 2
        max_m = max_i + S

        # S_partial[m] = sum_{k=0}^{m-1} (-1)^k / (2k+1)^2 mod p
        S_part = [F(0)] * (max_m + 2)
        for m in range(1, max_m + 2):
            k = m - 1
            S_part[m] = S_part[m-1] + F((-1)**k) / F(2*k+1)**2

        # Pi_i mod p for i = 0..max_i
        Pi = [F(1)] * (max_i + 1)
        for h in range(1, B+1):
            Pi[0] *= F(2*h+1)**2
        for i in range(max_i):
            Pi[i+1] = Pi[i] * F(2*(i+B+1)+1)**2 / F(2*(i+1)+1)**2

        # Pascal triangle up to N_total = max_i
        N_total = max_i
        binom = [[F(0)]*(N_total+1) for _ in range(N_total+1)]
        for n in range(N_total+1):
            binom[n][0] = F(1)
            if n > 0:
                binom[n][n] = F(1)
            for k in range(1, n):
                binom[n][k] = binom[n-1][k-1] + binom[n-1][k]

        # Build R_{a,j} mod p
        R = matrix(F, S+3, S)
        for a in range(S+3):
            n_a = a + 2*B
            for j in range(1, S+1):
                s = F(0)
                for i in range(n_a + 1):
                    m = i + j
                    Tm = F((-1)**m) * (G - S_part[m])
                    um = Tm / F(2*m+1)
                    s += F((-1)**i) * binom[n_a][i] * Pi[i] * um
                R[a, j-1] = s
        return R

    test_cases = [
        (5, 1), (5, 2), (5, 4),
        (10, 1), (10, 3), (10, 7),
        (15, 1), (15, 5), (15, 10),
        (20, 1), (20, 5), (20, 12),
        (25, 1), (25, 8), (25, 15),
        (30, 1), (30, 10), (30, 20),
    ]

    print(f"  Testing {len(test_cases)} (B,S) pairs, 3 random G each:")
    all_ok = True
    t0 = cputime()
    for (B, S) in test_cases:
        ranks = []
        for _ in range(3):
            G_val = random.randint(1, p-1)
            R = R_mod_p(B, S, G_val)
            ranks.append(R.rank())
        ok = all(r == S for r in ranks)
        mark = "OK" if ok else "FAIL"
        print(f"    B={B:3d}, S={S:3d}: ranks over 3 random G = {ranks}, "
              f"expected {S}  [{mark}]")
        if not ok:
            all_ok = False
    elapsed = cputime(t0)

    record("Rank theorem holds mod p for all test cases",
           all_ok,
           f"{len(test_cases)} cases, {3*len(test_cases)} total checks, "
           f"{elapsed:.2f} s")


module11_rank_modular()
